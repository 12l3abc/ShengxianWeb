# 倒计时特效关键代码（带详细注释）

## 一、限时秒杀倒计时系统

### 1. 核心倒计时类
```javascript
/**
 * 限时秒杀模块主类
 * 包含完整的倒计时功能、进度条、粘性显示等特效
 */
class FlashSale {
    constructor() {
        // 获取DOM元素引用
        this.productsContainer = document.getElementById('flash-sale-products'); // 商品容器
        this.hourElement = document.querySelector('.countdown-hour'); // 小时显示元素
        this.minuteElement = document.querySelector('.countdown-minute'); // 分钟显示元素
        this.secondElement = document.querySelector('.countdown-second'); // 秒显示元素
        this.timerInterval = null; // 倒计时定时器引用，用于后续清除
    }

    /**
     * 初始化限时秒杀模块
     * 整合所有特效功能
     */
    init() {
        this.startCountdown(); // 启动倒计时
        this.loadFlashSaleProducts(); // 加载秒杀商品
        this.setupCountdownProgress(); // 设置倒计时进度条
        this.enableTiltHover(); // 启用商品卡片倾斜效果
        this.injectFlameBar(); // 注入火焰进度条
        this.addStickyCountdown(); // 添加粘性倒计时
        this.addParticleExplosionEffect(); // 添加粒子爆炸特效
        this.addPriceNumberBounceEffect(); // 添加价格数字弹跳效果
    }

    /**
     * 开始倒计时
     * 核心的倒计时逻辑实现
     */
    startCountdown() {
        // 设置初始倒计时时间（1小时45分30秒）
        let hours = 1; // 小时
        let minutes = 45; // 分钟
        let seconds = 30; // 秒

        // 立即更新一次显示，避免初始延迟
        this.updateCountdownDisplay(hours, minutes, seconds);

        // 启动倒计时定时器，每秒执行一次
        this.timerInterval = setInterval(() => {
            seconds--; // 秒数减1

            // 时间递减逻辑处理
            if (seconds < 0) { // 秒数小于0时
                seconds = 59; // 秒数重置为59
                minutes--; // 分钟减1

                if (minutes < 0) { // 分钟小于0时
                    minutes = 59; // 分钟重置为59
                    hours--; // 小时减1

                    if (hours < 0) { // 小时小于0时（倒计时结束）
                        // 重新开始倒计时
                        hours = 1;
                        minutes = 45;
                        seconds = 30;
                        // 触发新一轮秒杀的逻辑
                        this.loadFlashSaleProducts();
                    }
                }
            }

            // 更新显示和进度条
            this.updateCountdownDisplay(hours, minutes, seconds); // 更新数字显示
            this.updateCountdownProgress(hours, minutes, seconds); // 更新进度条
        }, 1000); // 1000ms = 1秒
    }

    /**
     * 更新倒计时显示
     * @param {number} hours - 小时数
     * @param {number} minutes - 分钟数
     * @param {number} seconds - 秒数
     */
    updateCountdownDisplay(hours, minutes, seconds) {
        // 使用padStart方法确保两位数显示（如：1 → 01）
        if (this.hourElement) {
            this.hourElement.textContent = hours.toString().padStart(2, '0');
        }
        if (this.minuteElement) {
            this.minuteElement.textContent = minutes.toString().padStart(2, '0');
        }
        if (this.secondElement) {
            this.secondElement.textContent = seconds.toString().padStart(2, '0');
        }
    }

    /**
     * 设置倒计时进度条
     * 创建并配置视觉进度条元素
     */
    setupCountdownProgress() {
        // 创建进度条外层容器
        const bar = document.createElement('div');
        bar.className = 'countdown-progress'; // 添加类名
        bar.style.position = 'relative'; // 相对定位
        bar.style.height = '6px'; // 高度6px
        bar.style.background = 'rgba(255,255,255,0.15)'; // 半透明白色背景
        bar.style.borderRadius = '6px'; // 圆角6px
        bar.style.overflow = 'hidden'; // 隐藏溢出内容
        
        // 创建进度条内层元素（实际进度显示）
        const inner = document.createElement('div');
        inner.className = 'countdown-progress-inner'; // 添加类名
        inner.style.height = '100%'; // 高度100%
        inner.style.width = '100%'; // 初始宽度100%
        inner.style.background = 'linear-gradient(90deg,#ff9800,#ff5722)'; // 橙到红的渐变
        inner.style.transformOrigin = 'left center'; // 变换原点设为左侧中心
        inner.style.transition = 'transform 0.35s ease'; // 平滑过渡效果
        
        bar.appendChild(inner); // 将内层添加到外层容器
        
        // 获取要插入的容器元素
        const container = document.querySelector('.flash-sale-info') || document.querySelector('.flash-sale');
        if (container) {
            container.appendChild(bar); // 将进度条添加到容器
        }
        
        // 保存引用以便后续更新
        this.progressInner = inner;
        // 计算总秒数作为进度条的基准
        this.initialSeconds = 1 * 3600 + 45 * 60 + 30; // 总秒数 = 1小时 + 45分钟 + 30秒
    }

    /**
     * 更新倒计时进度条
     * @param {number} h - 当前小时
     * @param {number} m - 当前分钟
     * @param {number} s - 当前秒
     */
    updateCountdownProgress(h, m, s) {
        // 检查进度条元素和初始秒数是否存在
        if (!this.progressInner || !this.initialSeconds) return;
        
        // 计算剩余秒数
        const remain = Math.max(h * 3600 + m * 60 + s, 0); // 确保不小于0
        // 计算剩余时间比例
        const ratio = remain / this.initialSeconds;
        // 使用transform: scaleX()实现进度条动画效果
        this.progressInner.style.transform = `scaleX(${ratio})`;
    }

    /**
     * 添加粘性倒计时
     * 让倒计时始终保持在可视区域顶部
     */
    addStickyCountdown() {
        // 获取倒计时元素
        const countdown = document.querySelector('.flash-sale-countdown') || document.querySelector('.countdown');
        if (!countdown) return; // 如果没有找到元素，直接返回
        
        // 设置粘性定位样式
        countdown.style.position = 'sticky'; // 粘性定位
        countdown.style.top = '10px'; // 距离顶部10px
        countdown.style.zIndex = '1100'; // 设置高z-index确保显示在最上层
        countdown.style.backdropFilter = 'blur(6px)'; // 背景模糊效果
        countdown.style.boxShadow = '0 6px 20px rgba(0,0,0,0.15)'; // 阴影效果
        countdown.style.transition = 'all 0.3s ease'; // 平滑过渡效果
        
        // 添加滚动时的透明度变化效果
        let lastScrollTop = 0; // 记录上一次滚动位置
        const handleScroll = () => {
            // 获取当前滚动位置
            const scrollTop = window.pageYOffset || document.documentElement.scrollTop;
            
            // 向下滚动且超过100px时
            if (scrollTop > lastScrollTop && scrollTop > 100) {
                countdown.style.opacity = '0.85'; // 降低透明度
                countdown.style.transform = 'translateY(-5px)'; // 向上移动5px
            } else {
                countdown.style.opacity = '1'; // 恢复完全不透明
                countdown.style.transform = 'translateY(0)'; // 恢复原位置
            }
            lastScrollTop = scrollTop; // 更新上一次滚动位置
        };
        
        // 添加滚动事件监听器，passive:true提升性能
        window.addEventListener('scroll', handleScroll, { passive: true });
        
        // 添加鼠标悬停效果
        countdown.addEventListener('mouseenter', () => {
            countdown.style.transform = 'scale(1.02)'; // 放大1.02倍
            countdown.style.boxShadow = '0 8px 24px rgba(0,0,0,0.2)'; // 增强阴影
        });
        
        countdown.addEventListener('mouseleave', () => {
            countdown.style.transform = 'scale(1)'; // 恢复原大小
            countdown.style.boxShadow = '0 6px 20px rgba(0,0,0,0.15)'; // 恢复原阴影
        });
    }

    /**
     * 销毁倒计时实例
     * 清理资源，避免内存泄漏
     */
    destroy() {
        // 清除定时器，避免页面卸载后仍在运行
        if (this.timerInterval) {
            clearInterval(this.timerInterval);
            this.timerInterval = null;
        }
        // 其他清理逻辑...
    }
}
```

### 1. 快速响应倒计时类
```javascript
/**
 * 快速响应倒计时特效
 * 用于显示客服60秒响应承诺
 */
class QuickResponseCountdown {
    constructor() {
        this.countdownValue = 60; // 60秒响应承诺
        this.init(); // 初始化
    }
    
    init() {
        this.createPromiseDisplay(); // 创建响应承诺展示
        this.startCountdown(); // 启动倒计时
    }
    
    /**
     * 创建响应承诺展示元素
     */
    createPromiseDisplay() {
        // 创建容器元素
        const promiseContainer = document.createElement('div');
        promiseContainer.className = 'response-promise';
        
        // 设置容器样式
        promiseContainer.style.display = 'flex';
        promiseContainer.style.alignItems = 'center';
        promiseContainer.style.gap = '10px';
        promiseContainer.style.padding = '10px 15px';
        promiseContainer.style.backgroundColor = '#f0f9ff';
        promiseContainer.style.borderRadius = '8px';
        promiseContainer.style.border = '1px solid #bae6fd';
        
        // 创建图标
        const icon = document.createElement('div');
        icon.className = 'promise-icon';
        icon.innerHTML = '⏰';
        icon.style.fontSize = '20px';
        
        // 创建文本内容
        const text = document.createElement('div');
        text.className = 'promise-text';
        text.innerHTML = `人工客服 <span class="countdown-display">${this.countdownValue}</span> 秒内响应`;
        text.style.fontSize = '14px';
        text.style.color = '#0c4a6e';
        
        // 创建倒计时显示元素
        const countdownDisplay = text.querySelector('.countdown-display');
        countdownDisplay.style.fontWeight = 'bold';
        countdownDisplay.style.color = '#0284c7';
        countdownDisplay.style.margin = '0 3px';
        
        // 插入到页面指定位置
        const targetContainer = document.querySelector('.after-sales-header') || document.body;
        if (targetContainer) {
            targetContainer.appendChild(promiseContainer);
            promiseContainer.appendChild(icon);
            promiseContainer.appendChild(text);
        }
    }
    
    /**
     * 启动倒计时
     */
    startCountdown() {
        // 获取倒计时显示元素
        const countdownElement = document.querySelector('.countdown-display');
        if (countdownElement) {
            // 更新倒计时函数
            const updateCountdown = () => {
                // 循环倒计时：(60-1+60)%60 = 59, 然后递减到0后重新开始
                this.countdownValue = (this.countdownValue - 1 + 60) % 60;
                // 当倒计时到0时重置为60
                if (this.countdownValue === 0) this.countdownValue = 60;
                // 更新显示
                countdownElement.textContent = this.countdownValue;
            };
            
            // 每秒更新一次
            setInterval(updateCountdown, 1000);
        }
    }
}
```

## 三、保质期倒计时

### 1. 保质期倒计时类
```javascript
/**
 * 保质期倒计时特效
 * 用于显示商品的保质期剩余时间
 */
class ShelfLifeCountdown {
    constructor() {
        this.products = []; // 商品列表
        this.init(); // 初始化
    }
    
    init() {
        this.initializeShelfLife(); // 初始化保质期显示
        // 每分钟更新一次，保质期不需要秒级精度
        setInterval(() => this.updateShelfLife(), 60000);
    }
    
    /**
     * 初始化保质期显示
     */
    initializeShelfLife() {
        // 获取所有商品卡片
        const cards = document.querySelectorAll('.dairy-card, .bakery-card');
        cards.forEach((card, index) => {
            // 获取或创建保质期元素
            let expiryElement = card.querySelector('.expiry-date, .shelf-life');
            if (!expiryElement) {
                // 创建保质期显示元素
                const expiryBadge = document.createElement('div');
                expiryBadge.className = 'expiry-countdown';
                expiryBadge.textContent = `剩余${Math.floor(Math.random() * 7) + 1}天`; // 随机1-7天
                expiryBadge.style.position = 'absolute';
                expiryBadge.style.bottom = '10px';
                expiryBadge.style.right = '10px';
                expiryBadge.style.background = 'rgba(255,152,0,0.9)';
                expiryBadge.style.color = '#fff';
                expiryBadge.style.padding = '6px 12px';
                expiryBadge.style.borderRadius = '15px';
                expiryBadge.style.fontSize = '11px';
                expiryBadge.style.fontWeight = 'bold';
                expiryBadge.style.zIndex = '10';
                
                // 找到合适的容器添加元素
                const cardContent = card.querySelector('.card-content, .product-info');
                if (cardContent) {
                    cardContent.style.position = 'relative';
                    cardContent.appendChild(expiryBadge);
                    expiryElement = expiryBadge;
                }
            }
            
            // 如果成功创建或找到元素，添加到商品列表
            if (expiryElement) {
                this.products.push({
                    card: card,
                    expiryElement: expiryElement,
                    daysRemaining: parseInt(expiryElement.textContent.replace(/[^0-9]/g, '')) // 提取数字
                });
            }
        });
    }
    
    /**
     * 更新保质期显示
     */
    updateShelfLife() {
        this.products.forEach(product => {
            // 每天减1天
            product.daysRemaining--;
            
            if (product.daysRemaining <= 0) {
                // 过期处理
                product.expiryElement.textContent = '已过期';
                product.expiryElement.style.background = 'rgba(239,68,68,0.9)';
                // 添加过期动画
                product.expiryElement.style.animation = 'blink 1s infinite';
            } else {
                // 更新剩余天数
                product.expiryElement.textContent = `剩余${product.daysRemaining}天`;
                
                // 根据剩余天数改变颜色
                if (product.daysRemaining <= 3) {
                    // 临期商品（3天以内）
                    product.expiryElement.style.background = 'rgba(251,146,60,0.9)';
                } else if (product.daysRemaining <= 7) {
                    // 保质期较短（7天以内）
                    product.expiryElement.style.background = 'rgba(255,193,7,0.9)';
                } else {
                    // 保质期较长
                    product.expiryElement.style.background = 'rgba(16,185,129,0.9)';
                }
            }
        });
    }
}
```

## 四、核心技术点分析

### 1. 倒计时实现方式
- **setInterval**: 最常用的倒计时实现方式，简单直接，适合大多数场景
- **setTimeout**: 可以替代setInterval，避免累积误差，适合高精度要求的场景
- **requestAnimationFrame**: 用于视觉动画效果，利用浏览器的渲染机制，性能更好

### 2. 时间计算逻辑
- **时间递减**: 秒数减1，当秒数小于0时，分钟减1，秒数重置为59；同理处理分钟和小时
- **时间格式化**: 使用`padStart(2, '0')`确保两位数显示
- **剩余时间计算**: 总秒数 = 小时×3600 + 分钟×60 + 秒

### 3. 视觉效果增强
- **进度条动画**: 使用CSS `transform: scaleX()`实现平滑的进度条效果
- **粘性定位**: 结合`position: sticky`和`backdrop-filter: blur()`实现现代UI效果
- **响应式设计**: 适配不同屏幕尺寸和滚动位置
- **微交互**: 添加悬停效果、透明度变化等增强用户体验

### 4. 性能优化
- **事件委托**: 避免过多的事件监听器
- **被动事件监听器**: `{ passive: true }`提升滚动性能
- **资源清理**: 在组件销毁时清除定时器，避免内存泄漏
- **DOM缓存**: 缓存DOM元素引用，避免重复查找

### 5. 扩展性设计
- **模块化**: 类的设计便于扩展和维护
- **接口设计**: 清晰的方法命名和参数设计
- **钩子函数**: 在关键时间点提供扩展点（如倒计时结束时的回调）
